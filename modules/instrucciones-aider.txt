let _pfChartPosId = null;

window.pfShowChart = function(posId) {
  const pos = pfPositions.find(p => p.id === posId);
  if (!pos) return;
  _pfChartPosId = posId;

  // 1. Header
  const empresa = COMPANY_NAME[pos.ticker_ars] || COMPANY_NAME[pos.ticker_adr] || pos.ticker_ars;
  document.getElementById('pchTicker').textContent = pos.ticker_ars;
  document.getElementById('pchCompany').textContent = empresa;

  const pr = pfPrices[pos.ticker_ars];
  const preco = pr?.last != null ? pr.last : '—';
  const change = pr?.change != null ? pr.change : null;
  const changePct = pr?.changePercent != null ? pr.changePercent : null;
  
  document.getElementById('pchPrice').textContent = preco !== '—' ? '$'+Number(preco).toLocaleString('es-AR',{maximumFractionDigits:2}) : '—';
  if (change != null) {
    const s = change >= 0 ? '+' : '';
    const col = change >= 0 ? 'var(--green)' : 'var(--red)';
    document.getElementById('pchChange').textContent = s+'$'+Math.abs(change).toFixed(2);
    document.getElementById('pchChange').style.color = col;
  } else {
    document.getElementById('pchChange').textContent = '';
  }
  if (changePct != null) {
    const s = changePct >= 0 ? '+' : '';
    const col = changePct >= 0 ? 'var(--green)' : 'var(--red)';
    document.getElementById('pchChangePct').textContent = s+changePct.toFixed(2)+'%';
    document.getElementById('pchChangePct').style.color = col;
  } else {
    document.getElementById('pchChangePct').textContent = '';
  }

  // 2. Info bar
  const fmtArs = v => '$'+Number(v).toLocaleString('es-AR',{maximumFractionDigits:2});
  const fmtUsd = v => v != null ? 'U$S '+Math.round(v).toLocaleString('es-AR') : '—';
  document.getElementById('pch_entrada').textContent = fmtArs(pos.precio_compra_ars);
  document.getElementById('pch_cant').textContent = Number(pos.cantidad).toLocaleString('es-AR');
  document.getElementById('pch_fecha').textContent = pos.fecha_compra ? new Date(pos.fecha_compra+'T00:00:00').toLocaleDateString('es-AR',{day:'2-digit',month:'2-digit',year:'numeric'}) : '—';

  const esBono = ['BONO_SOBERANO','DOLAR_LINKED','BONO_CER','BONO_DUAL'].includes(pos.tipo);

  const usdInv = pos.mep_compra>0 ? (pos.precio_compra_ars*pos.cantidad)/(factor*pos.mep_compra) : null;
  const usdAct = (pfMep && preco != null && preco !== '—') ? (preco*pos.cantidad)/(factor*pfMep) : null;
  const gan    = (usdInv!=null && usdAct!=null) ? usdAct-usdInv : null;
  const pct    = (gan!=null && usdInv>0) ? gan/usdInv : null;
  
  document.getElementById('pch_invertido').textContent = fmtUsd(usdInv);
  document.getElementById('pch_actual').textContent = fmtUsd(usdAct);
  if (gan != null) {
    const s = gan >= 0 ? '+' : '';
    const col = gan >= 0 ? 'var(--green)' : 'var(--red)';
    document.getElementById('pch_ganancia').textContent = s+fmtUsd(gan);
    document.getElementById('pch_ganancia').style.color = col;
  } else {
    document.getElementById('pch_ganancia').textContent = '—';
    document.getElementById('pch_ganancia').style.color = '';

  const levelsEl = document.getElementById('pchLevels');
  let levelsHtml = '';

' ('+slDist.toFixed(2)+'%)' : ''}</span>`;

function getTvSymbol(pos) {
  // Para tickers ARS (terminan en .BA): directamente el ticker_ars
  // Para tickers ADR: usar NYSE:{ticker_adr}
  if (pos.ticker_adr) {
    // Si tiene ADR, usar ese para el chart
    return 'NYSE:' + pos.ticker_adr;
  }
  // Si no tiene ADR, asumir que es .BA
  return pos.ticker_ars + '.BA';
}

function buildTvWidget(pos) {
  const symbol = getTvSymbol(pos);
  const src = `https://s.tradingview.com/widgetembed/?symbol=${symbol}&interval=D&theme=dark&style=1&locale=es&hideideas=1&toolbarbg=131322&allow_symbol_change=0&saveimage=0&studies=[]&no_sidebar=1&details=0&hotlist=0&calendar=0&news=0&range=12M&condateranges=0&height=500`;
  
  const iframe = document.getElementById('pchTvFrame');
  iframe.src = src;

  // Overlay lines
  const overlay = document.getElementById('pchOverlay');
  overlay.innerHTML = '';

  // Estimar priceHigh / priceLow: precio actual ±30%
  const preco = pfPrices[pos.ticker_ars]?.last;
  if (preco == null) return;

  const range = preco * 0.30;
  const priceHigh = preco + range;
  const priceLow = preco - range;

  function calcTop(price) {
    return 100 - ((price - priceLow) / (priceHigh - priceLow) * 80 + 10);
  }

  function createLine(price, label, color) {
    const top = calcTop(price);
    if (top < 0 || top > 100) return;
    const line = document.createElement('div');
    line.style.cssText = `position:absolute;top:${top}%;left:0;right:0;height:1px;background:${color};border-top:1px dashed ${color};`;
    const badge = document.createElement('span');
    badge.id = 'pchOverlayLine';
    badge.style.cssText = `position:absolute;left:8px;top:-10px;background:${color};color:#fff;padding:1px 6px;border-radius:3px;font-family:'IBM Plex Mono',monospace;font-size:9px;font-weight:600;white-space:nowrap;`;
    badge.textContent = `${label} $${Number(price).toFixed(2)}`;
    line.appendChild(badge);
    overlay.appendChild(line);
  }

  // Entry (violeta)
  createLine(entryPrice, 'ENTRY', '#7c7cf0');
  // SL (rojo)
  if (pos.stop_loss) {
    createLine(pos.stop_loss, 'SL', '#ef5350');
  }
  // TP (verde)
  if (pos.take_profit) {
    createLine(pos.take_profit, 'TP', '#26a69a');
  }
}

function buildDetailGrid(pos) {
  const pr = pfPrices[pos.ticker_ars];
  const preco = pr?.last;
  const adr = pos.ticker_adr ? (pfAdrPrices[pos.ticker_adr]?.last) : null;

  const items = [];

  // Precio actual
  items.push({ label: 'Precio actual', value: preco != null ? '$'+Number(preco).toLocaleString('es-AR',{maximumFractionDigits:2}) : '—' });
  
  // Máx 3m (estimado: +15% del actual)
  const max3m = preco ? preco * 1.15 : null;
  items.push({ label: 'Máx 3m (est.)', value: max3m ? '$'+max3m.toFixed(2) : '—' });
  
  // Mín 3m (estimado: -15% del actual)
  const min3m = preco ? preco * 0.85 : null;
  items.push({ label: 'Mín 3m (est.)', value: min3m ? '$'+min3m.toFixed(2) : '—' });
  
  // SMA 50 (estimado simple de últimos datos si están disponibles)
  items.push({ label: 'SMA 50', value: '—' });
  
  // SMA 200
  items.push({ label: 'SMA 200', value: '—' });

  // RSI
  items.push({ label: 'RSI (14)', value: '—' });

  // Beta
  const posCalced = pfCalc(pos);
  const beta = posCalced?.betaAdr;
  items.push({ label: 'Beta', value: beta != null ? beta.toFixed(2) : '—' });

  // Sector (del SECTOR_MAP)
  const sector = window.SECTOR_MAP ? SECTOR_MAP[pos.ticker_ars] || SECTOR_MAP[pos.ticker_adr] || '—' : '—';
  items.push({ label: 'Sector', value: sector });

  // Dividendo (no tenemos datos)
  items.push({ label: 'Dividendo', value: '—' });

  return items.map(item => `
    <div style="background:#1a1a30;border-radius:6px;padding:8px 10px;display:flex;flex-direction:column;gap:2px">
      <span style="font-size:8px;text-transform:uppercase;color:#666;letter-spacing:0.3px;font-family:'IBM Plex Mono',monospace">${item.label}</span>
      <span style="font-size:12px;font-weight:600;color:#ddd;font-family:'IBM Plex Mono',monospace">${item.value}</span>
    </div>
  `).join('');
}

window.pfChartClose = function(event) {
  if (event && event.target && event.target.id !== 'pfChartOverlay') return;
  const overlay = document.getElementById('pfChartOverlay');
  if (overlay) {
    overlay.style.display = 'none';
    // Detener widget TV (liberar memoria)
    const iframe = document.getElementById('pchTvFrame');
    if (iframe) iframe.src = '';
    document.getElementById('pchOverlay').innerHTML = '';
  }
  _pfChartPosId = null;
};

// Create alert from chart modal
window.pfChartCreateAlert = function() {
  const posId = _pfChartPosId;
  if (!posId) return;
  const pos = pfPositions.find(p => p.id === posId);
  if (!pos) return;

  const pr = pfPrices[pos.ticker_ars];
  const currentPrice = pr?.last;

  // Open alert creation with pre-filled data
  // Dispatch custom event for alerts module
  try {
    const event = new CustomEvent('pfOpenAlert', {
      detail: {
        ticker: pos.ticker_adr || pos.ticker_ars,
        tickerArs: pos.ticker_ars,
        tickerAdr: pos.ticker_adr,
        currentPrice: currentPrice,
        sl: pos.stop_loss,
        tp: pos.take_profit,
      }
    });
    document.dispatchEvent(event);
    // Fallback: if no alerts module, try window function
    if (typeof window.alOpen === 'function') {
      const ticker = pos.ticker_adr || pos.ticker_ars;
      window.alOpen(ticker);
    } else if (typeof window.alModuleOpen === 'function') {
      window.alModuleOpen(pos.ticker_ars);
    }
  } catch(e) {
    console.warn('[pfChart] No se pudo abrir alerta:', e);
  }
};

// Edit position from chart modal
window.pfChartEditPos = function() {
  const posId = _pfChartPosId;
  if (!posId) return;
  pfChartClose();
  setTimeout(() => pfModalOpen(posId), 100);
};

// Escape key to close chart modal
document.addEventListener('keydown', function(e) {
  if (e.key === 'Escape') {
    const overlay = document.getElementById('pfChartOverlay');
    if (overlay && overlay.style.display === 'flex') {
      pfChartClose();
    }
  }
});

## NOTAS IMPORTANTES

1. El `COMPANY_NAME` map se agregó como constante global fuera de la IIFE existente. Esto es necesario porque las funciones `pfShowChart` también están fuera (Event listeners no pueden estar dentro de la IIFE para funciones llamadas por onclick).

2. La función `pfChartClose`, `pfShowChart`, `pfChartCreateAlert`, `pfChartEditPos` son todas funciones globales (window.*) para que puedan llamarse desde `onclick` en HTML.

3. `getTvSymbol()` construye el símbolo para TradingView: si tiene ADR usa `NYSE:{adr}`, si no usa `{ticker_ars}.BA`.

4. Las líneas overlay SL/TP/Entry usan el cálculo de posición vertical: `top% = 100 - ((price - priceLow) / (priceHigh - priceLow) * 80 + 10)`.

5. Los indicadores SMA, RSI, dividendo se muestran como "—" porque no tenemos datos históricos disponibles en el módulo portafolio.

6. El widget TV usa `no_sidebar=1&details=0&hotlist=0&calendar=0&news=0` para máxima limpieza.

7. Al cerrar, se detiene el iframe cambiando src a '' para liberar memoria.

8. La tecla Escape está manejada con un `document.addEventListener('keydown', ...)`.

9. El botón "Crear alerta desde aquí" intenta disparar un CustomEvent 'pfOpenAlert' para que el módulo de alertas lo escuche. Como fallback, intenta llamar a `window.alOpen()` o `window.alModuleOpen()` si existen.
